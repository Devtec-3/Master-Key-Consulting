## STEP 3 — FIRESTORE DATABASE SCHEMA

### Collection: `reviews`

```
reviews/{reviewId}
  - name: string         // Client name
  - location: string     // e.g. "Abuja, FCT"
  - rating: number       // 1–5
  - comment: string      // Review text
  - service: string      // Which service they used
  - approved: boolean    // Admin must approve before showing publicly
  - createdAt: timestamp
```

### Collection: `projects` (Portfolio)

```
projects/{projectId}
  - title: string
  - location: string
  - serviceType: string  // e.g. "Borehole Drilling"
  - description: string
  - date: string         // e.g. "March 2025"
  - imageURL: string     // Firebase Storage URL
  - featured: boolean
  - createdAt: timestamp
```

### Collection: `blog`

```
blog/{postId}
  - title: string
  - slug: string         // URL-friendly e.g. "survey-at-ekiti-mine"
  - content: string      // Rich text / Markdown
  - coverImageURL: string
  - tags: array          // e.g. ["mineral exploration", "kwara"]
  - published: boolean
  - createdAt: timestamp
```

### Collection: `bookings`

```
bookings/{bookingId}
  - fullName: string
  - email: string
  - phone: string
  - location: string     // Where they want the work done
  - serviceType: string
  - preferredDate: string
  - message: string      // Details about the job
  - status: string       // "new" | "contacted" | "completed"
  - createdAt: timestamp
```

### 📄 BookingPage.jsx

**This is the most important page.**

```jsx
import emailjs from "emailjs-com";
import { collection, addDoc, serverTimestamp } from "firebase/firestore";

const handleSubmit = async (e) => {
  e.preventDefault();
  setLoading(true);
  try {
    // 1. Save to Firestore
    await addDoc(collection(db, "bookings"), {
      ...formData,
      status: "new",
      createdAt: serverTimestamp(),
    });

    // 2. Send email notification via EmailJS
    await emailjs.send(
      import.meta.env.VITE_EMAILJS_SERVICE_ID,
      import.meta.env.VITE_EMAILJS_TEMPLATE_ID,
      {
        from_name: formData.fullName,
        from_email: formData.email,
        phone: formData.phone,
        location: formData.location,
        service_type: formData.serviceType,
        preferred_date: formData.preferredDate,
        message: formData.message,
      },
      import.meta.env.VITE_EMAILJS_PUBLIC_KEY
    );

    toast.success("Booking submitted! We will contact you shortly.");
    resetForm();
  } catch (err) {
    toast.error("Something went wrong. Please try again.");
  } finally {
    setLoading(false);
  }
};
```

## STEP 10 — FIREBASE FIRESTORE SECURITY RULES

```js
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Public can READ approved reviews and published projects/blog
    match /reviews/{id} {
      allow read: if resource.data.approved == true;
      allow create: if true; // anyone can submit a review
      allow update, delete: if request.auth != null;
    }

    match /projects/{id} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /blog/{id} {
      allow read: if resource.data.published == true;
      allow write: if request.auth != null;
    }

    match /bookings/{id} {
      allow create: if true; // public can submit bookings
      allow read, update, delete: if request.auth != null;
    }
  }
}
```
